56def _http_error(exc: Exception) -> HTTPException: 

57 if isinstance(exc, (ListingNotFoundError, ComplaintNotFoundError)): 

58 code = status.HTTP_404_NOT_FOUND 

59 elif isinstance(exc, ModerationForbiddenError): 

60 code = status.HTTP_403_FORBIDDEN 

61 elif isinstance(exc, (DuplicateComplaintError, ComplaintConflictError)): 

62 code = status.HTTP_409_CONFLICT 

63 else: 

64 code = status.HTTP_422_UNPROCESSABLE_CONTENT 

65 return HTTPException(status_code=code, detail=str(exc))
