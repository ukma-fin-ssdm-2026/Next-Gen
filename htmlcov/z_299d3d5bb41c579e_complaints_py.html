34def normalize_complaint_text(value: str | None, max_length: int, label: str) -> str: 

35 """Обрізає пробіли й перевіряє максимальну довжину текстового поля.""" 

36 text = (value or "").strip() 

37 if len(text) > max_length: 

38 raise InvalidComplaintError( 

39 f"{label} не може перевищувати {max_length} символів" 

40 ) 

41 return text
