52def _custom_openapi() -> dict: 

53 """Прибирає автоматичний 422 з операцій, які не мають що валідувати. 

54 

55 Якщо операція приймає лише cookie сесії (без тіла, path, query чи header), 

56 FastAPI все одно додає `422 HTTPValidationError`, хоча ендпоїнт його не 

57 повертає. Явно задекларовані 422 (власна схема помилки) не чіпаються. 

58 """ 

59 if app.openapi_schema: 

60 return app.openapi_schema 

61 schema = get_openapi( 

62 title=app.title, 

63 version=app.version, 

64 description=app.description, 

65 routes=app.routes, 

66 ) 

67 for path_item in schema.get("paths", {}).values(): 

68 for operation in path_item.values(): 

69 if not isinstance(operation, dict) or "requestBody" in operation: 

70 continue 

71 params = operation.get("parameters", []) 

72 response_422 = operation.get("responses", {}).get("422", {}) 

73 auto = "HTTPValidationError" in str(response_422) 

74 if auto and all(item.get("in") == "cookie" for item in params): 

75 operation["responses"].pop("422") 

76 app.openapi_schema = schema 

77 return schema

149def resolve_frontend_file(frontend_path: str, frontend_root: Path) -> Path: 

150 """Повертає файл React build-у для шляху або index.html для SPA-маршруту. 

151 

152 404 для зарезервованих префіксів backend, для шляхів поза build-ом і для 

153 невідомих file-like шляхів (з розширенням): вони не маскуються під 

154 успішний React-маршрут (TM-03). 

155 """ 

156 if frontend_path.startswith(RESERVED_FRONTEND_PREFIXES): 

157 raise HTTPException(status_code=status.HTTP_404_NOT_FOUND) 

158 

159 root = frontend_root.resolve() 

160 requested_file = (root / frontend_path).resolve() 

161 if ( 

162 frontend_path 

163 and requested_file.is_relative_to(root) 

164 and requested_file.is_file() 

165 ): 

166 return requested_file 

167 if PurePosixPath(frontend_path).suffix: 

168 raise HTTPException(status_code=status.HTTP_404_NOT_FOUND) 

169 return root / "index.html" 

170
